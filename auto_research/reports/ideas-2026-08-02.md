from scipy.optimize import curve_fit
import numpy as np

def saturation(n, P_max, P0, lam):
    return P_max - (P_max - P0) * np.exp(-lam * n)

# Data: cycles [0, 1_corrected, 2, 3], weights [1.0, 0.5, 1.0, 1.0]
cycles = np.array([0, 1, 2, 3])

P_max, P0, lam = popt
ci_95 = 1.96 * np.sqrt(np.diag(pcov))
